% Figure 8 draft A: the 32 spin states as one vertical strip partitioned 12 | 4 | 16; the three spatial level rules
% at left joined to their partner block: solid for even parity, dashed for odd. Weights read as block heights.
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{numbers.tex}
\begin{document}
\begin{tikzpicture}[plate,even/.style={line width=.7pt,draw=PlateInk},odd/.style={line width=.7pt,draw=PlateInk,dash pattern=on 2.6pt off 1.6pt}]
\path[use as bounding box] (0,0) rectangle (15,9.6);
\pgfmathsetmacro{\u}{.24}
% the spin space as a strip of 32 cells, bottom to top: A1 (12), A2 (4), E (16 as 8 pairs)
\begin{scope}[shift={(10.2,.9)}]
  \foreach \k in {0,...,31} {\draw[line width=.3pt,draw=PlateInk!60] (0,{\u*\k}) rectangle (.9,{\u*(\k+1)});}
  \draw[heavy] (0,0) rectangle (.9,{\u*32});
  \draw[heavy] (0,{\u*12})--(.9,{\u*12}); \draw[heavy] (0,{\u*16})--(.9,{\u*16});
  \foreach \k in {18,20,...,30} {\draw[line width=.6pt,draw=PlateInk] (0,{\u*\k})--(.9,{\u*\k});}
  \node[lab,anchor=west] at (1.1,{\u*6}) {$\mathrm A_1$, $\weightEvenAone$};
  \node[lab,anchor=west] at (1.1,{\u*14}) {$\mathrm A_2$, $\weightEvenAtwo$};
  \node[lab,anchor=west] at (1.1,{\u*24}) {$\mathrm E$, $\weightEvenE$ pairs};
  \node[sub,anchor=south] at (.45,{\u*32+.12}) {$\spin$, $\spinDim$ states};
  \coordinate (bA) at (0,{\u*6}); \coordinate (bB) at (0,{\u*14}); \coordinate (bE) at (0,{\u*24});
\end{scope}
% spatial species as level rules
\draw[heavy] (2.2,7.0)--(4.4,7.0); \node[lab,anchor=south west] at (2.2,7.05) {$\mathrm A_1$};
\draw[heavy] (2.2,4.6)--(4.4,4.6); \node[lab,anchor=south west] at (2.2,4.65) {$\mathrm A_2$};
\draw[heavy] (2.2,2.2)--(4.4,2.2) (2.2,2.28)--(4.4,2.28); \node[lab,anchor=south west] at (2.2,2.33) {$\mathrm E$};
\node[sub,anchor=south] at (3.3,8.2) {spatial species};
% the pairings
\draw[even] (4.4,7.0)--(bA); \draw[odd] (4.4,7.0)--(bB);
\draw[even] (4.4,4.6)--(bB); \draw[odd] (4.4,4.6)--(bA);
\draw[even] (4.4,2.24)--($(bE)+(0,.05)$); \draw[odd] (4.4,2.24)--($(bE)+(0,-.05)$);
% a two-item key
\draw[even] (2.2,.9)--(3.2,.9); \node[sub,anchor=west] at (3.3,.9) {even parity, $\chi_+$};
\draw[odd] (5.6,.9)--(6.6,.9); \node[sub,anchor=west] at (6.7,.9) {odd parity, $\chi_-$};
\end{tikzpicture}
\end{document}
