% Study E. The Harter idiom applied: two ink weights, solid black crescent shading, dashed hidden lines and ghosts,
% regular hatch for walls and cut faces, flat gouache under ink. No grey linework anywhere.

\tikzset{heavy/.style={line width=1.0pt,draw=PlateInk},fine/.style={line width=.4pt,draw=PlateInk},
         hidden/.style={line width=.4pt,draw=PlateInk,dash pattern=on 1.6pt off 1.3pt},
         ghost/.style={line width=.5pt,draw=PlateInk,dash pattern=on 1.8pt off 1.4pt},
         harrow/.style={heavy,-{Stealth[length=2.4mm,width=2.0mm]}}}
% atom: flat colour, ink contour, solid black crescent on the lower right (light from the upper left)
\newcommand\hatom[4]{% x y r fill
  \fill[#4] ({#1},{#2}) circle[radius=#3];
  \begin{scope}\clip ({#1},{#2}) circle[radius=#3];
    \fill[PlateInk,even odd rule] ({(#1)+0.10*#3},{(#2)-0.10*#3}) circle[radius=#3] ({(#1)-0.32*#3},{(#2)+0.32*#3}) circle[radius={0.95*#3}];
  \end{scope}
  \draw[line width=.6pt,draw=PlateInk] ({#1},{#2}) circle[radius=#3];}
\begin{document}
\begin{tikzpicture}[plate]
\path[use as bounding box] (0,0) rectangle (15,13.2);
\node[panel] at (0,13.15) {Study E. Atoms: flat gouache, ink contour, solid black crescent};
\foreach \x/\col/\lab in {1.2/white/{H},3.0/PlateInk/{C},4.8/ColNitrogen/{N},6.6/ColOxygen/{O},8.4/ColPotassium/{K},10.2/ColRubidium/{Rb}} {
  \hatom{\x}{11.6}{.42}{\col}
  \node[small] at (\x,10.85) {\lab};
}
\node[note,anchor=west,text width=3.2cm,align=left] at (11.4,11.6) {carbon is ink-filled, so it carries no crescent; its contour and label do the work};

\node[panel] at (0,10.3) {Methylamine with rod bonds and heavy outer contours; labels beside, in ink};
\begin{scope}[shift={(3.3,8.1)}]
  \renewcommand\molbond[4]{\draw[line width=.5pt,draw=PlateInk,double distance=2.4pt,double=white] (#1,#2)--(#3,#4);}
  \renewcommand\molatom[4]{%
    \if#1H\hatom{#3}{#4}{.19}{white}\else\if#1C\hatom{#3}{#4}{.29}{PlateInk}\else\hatom{#3}{#4}{.29}{ColNitrogen}\fi\fi
    \if#1H\node[font=\fontsize{8}{9}\selectfont,inner sep=.6pt] at ($(#3,#4)!-.5cm!(0,0)$) {$\mathrm{#1}_{#2}$};\else
      \node[font=\fontsize{8}{9}\selectfont,inner sep=.6pt,anchor=north] at ($(#3,#4)+(0,-.36)$) {$\mathrm{#1}_{#2}$};\fi}
  \pic[scale=1.0] at (0,0) {mla-ref};
\end{scope}
% ghost: the same molecule as a dashed ghost, for a version or a rotated copy
\begin{scope}[shift={(9.6,8.1)}]
  \renewcommand\molbond[4]{\draw[ghost] (#1,#2)--(#3,#4);}
  \renewcommand\molatom[4]{\if#1H\draw[ghost,fill=white] (#3,#4) circle[radius=.19];\else\draw[ghost,fill=white] (#3,#4) circle[radius=.29];\fi}

\node[panel] at (0,5.9) {Thick tube in the same idiom: heavy silhouette, dashed hidden rim, black shadow band, hatched cut faces};
\begin{scope}[shift={(4.2,3.1)}]
  \def\Ro{2.5}\def\Ri{1.75}\def\Rm{2.125}\def\Hh{.6}\def\E{.45}
  % far inner wall through the opening: hatched (it is a wall in shadow)
  \begin{scope}\clip (0,\Hh) ellipse[x radius=\Ri,y radius={\E*\Ri}];
    \path[fill=white] (-\Ri,\Hh) arc[start angle=180,end angle=0,x radius=\Ri,y radius={\E*\Ri}] -- (\Ri,-\Hh) arc[start angle=0,end angle=180,x radius=\Ri,y radius={\E*\Ri}] -- cycle;
    \foreach \x in {-3.0,-2.75,...,3.0} \draw[fine] (\x,-2)--({\x+1.6},2);
  \end{scope}
  % outer wall with wedge removed
  \foreach \ta/\tb in {-90/-80,-5/90} {
    \path[fill=ColNitrogen!18] plot[domain=\ta:\tb,samples=25,variable=\a] ({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)}) -- plot[domain=\tb:\ta,samples=25,variable=\a] ({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)}) -- cycle;
  }
  % solid black shadow band on the right of the wall (Harter crescent, bent around the cylinder)
  \path[fill=PlateInk] plot[domain=48:90,samples=20,variable=\a] ({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)}) -- plot[domain=90:48,samples=20,variable=\a] ({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)}) -- cycle;
  \path[fill=ColNitrogen!18] plot[domain=48:76,samples=20,variable=\a] ({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)-.08}) -- plot[domain=76:48,samples=20,variable=\a] ({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)+.08}) -- cycle;
  \draw[heavy] plot[domain=-5:90,samples=30,variable=\a] ({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)});
  \draw[heavy] plot[domain=-90:-80,samples=6,variable=\a] ({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)});
  \draw[heavy] (-\Ro,-\Hh)--(-\Ro,\Hh) (\Ro,-\Hh)--(\Ro,\Hh);
  % ghost band in the wedge and the tu path
  \draw[hidden] plot[domain=-80:-5,samples=25,variable=\a] ({\Rm*sin(\a)},{\Hh-\E*\Rm*cos(\a)});
  \draw[hidden] plot[domain=-80:-5,samples=25,variable=\a] ({\Rm*sin(\a)},{-\Hh-\E*\Rm*cos(\a)});
  \draw[ColOxygen,line width=1.0pt,-{Stealth[length=2.2mm,width=1.8mm]}] plot[domain=0:1,samples=30,variable=\s] ({\Rm*sin(-60*\s)},{\Hh-2*\Hh*\s-\E*\Rm*cos(60*\s)});
  % cut faces hatched
  \foreach \a in {-80,-5} {
    \path[fill=white] ({\Ri*sin(\a)},{\Hh-\E*\Ri*cos(\a)})--({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)})--({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)})--({\Ri*sin(\a)},{-\Hh-\E*\Ri*cos(\a)})--cycle;
    \begin{scope}\clip ({\Ri*sin(\a)},{\Hh-\E*\Ri*cos(\a)})--({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)})--({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)})--({\Ri*sin(\a)},{-\Hh-\E*\Ri*cos(\a)})--cycle;
      \foreach \y in {-3,-2.8,...,2} \draw[fine] (-3.5,\y)--(3.5,{\y+1.6});\end{scope}
    \draw[heavy] ({\Ri*sin(\a)},{\Hh-\E*\Ri*cos(\a)})--({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)})--({\Ro*sin(\a)},{-\Hh-\E*\Ro*cos(\a)})--({\Ri*sin(\a)},{-\Hh-\E*\Ri*cos(\a)})--cycle;
    \draw[ColNitrogen,line width=1.0pt] ({\Rm*sin(\a)},{\Hh-\E*\Rm*cos(\a)})--({\Rm*sin(\a)},{-\Hh-\E*\Rm*cos(\a)});
  }
  % top face
  \path[fill=white] plot[domain=-5:280,samples=80,variable=\a] ({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)}) -- plot[domain=280:-5,samples=80,variable=\a] ({\Ri*sin(\a)},{\Hh-\E*\Ri*cos(\a)}) -- cycle;
  \draw[heavy] plot[domain=-5:280,samples=80,variable=\a] ({\Ro*sin(\a)},{\Hh-\E*\Ro*cos(\a)});
  \draw[heavy] plot[domain=-5:280,samples=80,variable=\a] ({\Ri*sin(\a)},{\Hh-\E*\Ri*cos(\a)});
  \draw[hidden] plot[domain=-5:280,samples=80,variable=\a] ({\Rm*sin(\a)},{\Hh-\E*\Rm*cos(\a)});
  \draw[ColNitrogen,line width=1.0pt,-{Stealth[length=2.2mm,width=1.8mm]}] plot[domain=0:118,samples=30,variable=\a] ({\Rm*sin(\a)},{\Hh-\E*\Rm*cos(\a)});
  \foreach \a in {0,120,240} \hatom{\Rm*sin(\a)}{\Hh-\E*\Rm*cos(\a)}{.09}{ColGold}
  \hatom{\Rm*sin(-60)}{-\Hh-\E*\Rm*cos(-60)}{.09}{ColGold}
  \draw[fine] ({\Rm*sin(0)},{\Hh-\E*\Rm*cos(0)})--(.9,-.05) node[anchor=west,inner sep=1pt,small] {$H$};
  \draw[fine] ({\Rm*sin(120)},{\Hh-\E*\Rm*cos(120)})--(2.8,1.9) node[anchor=west,inner sep=1pt,small] {$tH$};
  \draw[fine] ({\Rm*sin(240)},{\Hh-\E*\Rm*cos(240)})--(-2.8,1.9) node[anchor=east,inner sep=1pt,small] {$t^2H$};
  \draw[fine] ({\Rm*sin(-60)},{-\Hh-\E*\Rm*cos(-60)})--(-3.1,-1.45) node[anchor=east,inner sep=1pt,small] {$tuH$};
  \node[small] at (1.15,1.3) {$t$};
  \node[small] at (-1.75,-.25) {$tu$};
  \draw[harrow] (3.6,-1.3)--(3.6,1.3); \node[small,anchor=west] at (3.7,0) {$\eta$};
\end{scope}
\node[note,anchor=north west,text width=5.2cm,align=left] at (9.6,5.3) {heavy 1.0 pt for silhouettes and section edges, fine 0.4 pt for hatch, leaders and hidden lines (dashed); the shadow band is solid ink, bent around the wall; cut faces are hatched like Harter's walls; the band at mid-thickness is dashed where it is inside the solid};
