% primitives.tex, the drawing vocabulary of the plates. Load after figurestyle.sty.
% A line is ink (figurestyle.sty says where a hue or the gray takes over), at one of four weights, heavy 1pt
% (silhouettes, retained structure, action paths), fine .5pt (construction, detail), thin .25pt (leaders, axes behind a
% molecule) and hatch .25pt. Disc and node contours are .5pt.
% Actions are told apart by line style at action weight, solid for t, dashed for u or tu and beaded dots for a starred
% operation (b, E*). Arrowheads appear only where a path has a direction. A head stops about 3pt short of the dot it
% points at, and an arrow sits centered in its gap. The hidden part of a contour is the same stroke dashed (back), and
% a hidden or faded object's outline is dashed (ghost). A hidden point (every edge at it hidden) is pale in a ghost
% outline under the front lines, and a projection or lift between levels is dotted (projection). The one solid faded
% outline is figure 10's ghost, the undisplaced shape in gray lines on white. A receded object (grayed out, or in the
% background) is drawn in the gray, never thinner than .5pt, with pale gray fills.
% Leaders start on the object, stop short of the label and cross nothing where avoidable, with no halos. A sphere is
% lit from the upper left, its shadow at the lower right the deep tone of its hue, or pale gray for a white sphere.
% Version points are spheres of radius 2.5pt. Rods start where the bond leaves the far sphere. Hatch marks cut material.
\tikzset{
  heavy/.style={line width=1pt,draw=PlateInk},
  fine/.style={line width=.5pt,draw=PlateInk},
  thin/.style={line width=.25pt,draw=PlateInk},   % leaders, axes behind a molecule, chart loci, zero lines, hairlines
  hatch/.style={line width=.25pt,draw=PlateInk},  % cut material
  ring/.style={line width=.5pt,draw=PlateInk},    % the contour of a disc or a node
  faint/.style={fine,draw=PlateGray},             % a line of a background object (the subgroups outside an interval)
  leader/.style={thin},
  back/.style={dash pattern=on 1.5pt off 1pt},   % the hidden part of a contour: the same stroke as its front part, dashed
  ghost/.style={ring,dash pattern=on 1.25pt off 1pt},   % a hidden or faded object's outline, dashed
  projection/.style={line width=.5pt,draw=PlateInk,dash pattern=on 0pt off 1.5pt},  % dotted: a projection or lift between levels
  % actions: solid = t, dashed = u or tu, beaded dots = a starred operation (b, E*); the beads take a wider pen to read
  tlift/.style={heavy},
  ulift/.style={heavy,dash pattern=on 2.5pt off 1.5pt},
  blift/.style={line width=1.25pt,draw=PlateInk,dash pattern=on 0pt off 2.25pt},
  % arrows: the plate's head (>) on heavy strokes; smaller heads on lighter strokes
  tpath/.style={tlift,->},
  upath/.style={ulift,->},
  bpath/.style={blift,->},
  harrow/.style={heavy,->},                        % a vector, or a map from one picture to the next
  sarrow/.style={line width=.75pt,draw=PlateInk,-{Stealth[length=4.5pt,width=3.75pt]}},   % a mark on a glyph: spin, turning sense, displacement
  farrow/.style={fine,-{Stealth[length=4pt,width=3.25pt]}},   % a correspondence between two rows (figure 3's Psi)
  lab/.style={font=\fontsize{9}{11}\selectfont,inner sep=1pt},
  sub/.style={font=\fontsize{8}{10}\selectfont,inner sep=1pt,text=PlateInk},
  dense/.style={font=\fontsize{7}{8}\selectfont,inner sep=0pt},     % text in a small cell or graph node
  digit/.style={font=\fontsize{6.5}{7}\selectfont,inner sep=0pt},   % a digit in a disc
  under/.style={sub,anchor=north,align=center},                     % a note centered under its object
}
% ---- atom: \hatom{x}{y}{r}{lit}{shadow}, a sphere in an ink contour, lit from the upper left of the page whatever the
% pic's rotation, its shadow a crescent at the lower right up to .28 of the radius wide, and at least 1pt
\newcommand\hatom[5]{%
  \pgfgettransformentries{\hta}{\htb}{\htc}{\htd}{\hte}{\htf}%
  \pgfmathsetmacro{\htl}{135-atan2(\htb,\hta)}%   the page's upper left in the picture's frame
  \fill[#5] ({#1},{#2}) circle[radius=#3];
  \pgfmathsetmacro{\hts}{max(.14*(#3),.0176)}%   the shadow at least 1pt wide, so it shows past the ring on small spheres
  \fill[#4] ({(#1)+\hts*cos(\htl)},{(#2)+\hts*sin(\htl)}) circle[radius={(#3)-\hts}];
  \draw[ring] ({#1},{#2}) circle[radius=#3];}
% ---- rod: a cylinder in ink. \molrodj{px}{py}{qx}{qy}{dx}{dy}{uz}{r}: far center p, near center q, page unit
% direction (dx,dy) from p to q, uz = depth component of the bond direction (0 = in the page, 1 = at the viewer),
% r = radius of the far sphere. The rod starts on the junction circle where the cylinder leaves the far sphere
% (projected: a half ellipse bulging toward p), runs to q and is hidden there by the near sphere, painted later.
% White band, fine ink edges, the shadow edge (away from the upper-left light) heavier.
\def\rodw{.05}
\tikzset{rodfill/.style={fill=PlateWhite},rodedge/.style={line width=.5pt,draw=PlateInk},rodshade/.style={line width=.75pt,draw=PlateInk}}
\newcommand\molrodj[8]{%
  \pgfmathsetmacro{\rs}{sqrt((#8)*(#8)-\rodw*\rodw)}%
  \pgfmathsetmacro{\ex}{#1+\rs*(#5)}\pgfmathsetmacro{\ey}{#2+\rs*(#6)}%
  \pgfmathsetmacro{\nx}{-(#6)}\pgfmathsetmacro{\ny}{#5}\pgfmathsetmacro{\ea}{\rodw*(#7)}%
  \pgfmathsetmacro{\sg}{(\ny-\nx)<0 ? 1 : -1}%
  \path[rodfill] ({\ex+\rodw*\nx},{\ey+\rodw*\ny}) \foreach \ph in {105,120,...,255} {-- ({\ex+\ea*cos(\ph)*(#5)+\rodw*sin(\ph)*\nx},{\ey+\ea*cos(\ph)*(#6)+\rodw*sin(\ph)*\ny})}
     -- ({\ex-\rodw*\nx},{\ey-\rodw*\ny}) -- ({#3-\rodw*\nx},{#4-\rodw*\ny}) -- ({#3+\rodw*\nx},{#4+\rodw*\ny}) -- cycle;
  \draw[rodedge] ({\ex+\rodw*\nx},{\ey+\rodw*\ny}) \foreach \ph in {105,120,...,255} {-- ({\ex+\ea*cos(\ph)*(#5)+\rodw*sin(\ph)*\nx},{\ey+\ea*cos(\ph)*(#6)+\rodw*sin(\ph)*\ny})} -- ({\ex-\rodw*\nx},{\ey-\rodw*\ny});
  \ifdim\sg pt>0pt
    \draw[rodshade] ({\ex+\rodw*\nx},{\ey+\rodw*\ny}) -- ({#3+\rodw*\nx},{#4+\rodw*\ny});
    \draw[rodedge] ({\ex-\rodw*\nx},{\ey-\rodw*\ny}) -- ({#3-\rodw*\nx},{#4-\rodw*\ny});
  \else
    \draw[rodedge] ({\ex+\rodw*\nx},{\ey+\rodw*\ny}) -- ({#3+\rodw*\nx},{#4+\rodw*\ny});
    \draw[rodshade] ({\ex-\rodw*\nx},{\ey-\rodw*\ny}) -- ({#3-\rodw*\nx},{#4-\rodw*\ny});
  \fi}

% ---- Newman projection macros (used by the newman-* pics; coordinates in angstrom, pic scale sets the size)
% back carbon: a circle of radius \nmR with its hydrogens on ink bonds from the rim; front nitrogen: a disc at the
% center with its hydrogens on rod bonds. Order of painting: back bonds and atoms, circle, front bonds and atoms, center.
% The radii are in pt as printed at figure 6's pic scale .9: carbon 12.5pt, hydrogen 5pt, nitrogen 7.5pt.
\def\nmR{12.5pt/.9}\def\nmr{5pt/.9}
\newcommand\nmback[3]{\pgfmathsetmacro{\nml}{sqrt((#1)*(#1)+(#2)*(#2))}
  \draw[heavy] ({\nmR*(#1)/\nml},{\nmR*(#2)/\nml})--(#1,#2);
  \draw[ring,fill=PlateWhite] (#1,#2) circle[radius=\nmr];\node[digit] at (#1,#2) {#3};}
\newcommand\nmcircle{\draw[ring,fill=PlateGray] (0,0) circle[radius=\nmR];}
\newcommand\nmfront[3]{\draw[fine,double distance=2.5pt,double=PlateWhite] (0,0)--(#1,#2);
  \draw[ring,fill=PlateWhite] (#1,#2) circle[radius=\nmr];\node[digit] at (#1,#2) {#3};}
\newcommand\nmcenter{\draw[ring,fill=ColPaleBlue] (0,0) circle[radius=7.5pt/.9];}
