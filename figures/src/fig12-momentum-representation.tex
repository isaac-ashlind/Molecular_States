% Figure 12. Momentum representation. What the reader must see: the same states carry mode labels (J, M, K, m, p);
% K is the body-axis component of J, shown as latitude rings on the sphere |J| = sqrt(J(J+1)) (after Harter (1993)
% fig. 5.5.3, no energies); the seam turn shears the (m, K) lattice of torsional labels to kappa = m + rho K;
% periodic torsion modes drawn as radial plots on the torsion circle (the circle of figure 9) sort into A1, A2
% and E pairs, with level-rule typography (after fig. 4.2.3).
% Colours: gold = the shear line kappa = rho K; red = the version marks on the torsion circle; ink elsewhere. Hatch unused.
% Text reduced to labels (2026-10-04). rho = 0.3 is illustrative (declared);
% the species table and kappa rows are computed (compute/, checks/). Chosen from studies/alternatives/alt-fig12.

% ------------------------------------------------ A. the K ladder on the sphere |J| = sqrt(J(J+1)), J = 2
\begin{scope}[shift={(2.3,9.25)}]
  \pgfmathsetmacro{\s}{.56}\pgfmathsetmacro{\R}{sqrt(6)*\s}\pgfmathsetmacro{\E}{.32}

  \foreach \K in {2,1,0,-1,-2} {\pgfmathsetmacro{\r}{sqrt(6-\K*\K)*\s}\pgfmathsetmacro{\h}{\K*\s}

    \draw[fine] ({-\r},\h) arc[start angle=180,end angle=360,x radius=\r,y radius={\E*\r}];
    \draw[fine] ({\R+.05},\h)--({\R+.2},\h); \node[sub,anchor=west] at ({\R+.22},\h) {$K=\K$};}

  \node[lab,anchor=north west,inner sep=1pt] at ({sqrt(5)*\s*cos(-40)+.02},{\s+\E*sqrt(5)*\s*sin(-40)-.02}) {$\mathbf J$};
  \node[sub,anchor=north] at (0,{-\R-.3}) {$J=2$, $|\mathbf J|=\sqrt{J(J+1)}$};

\foreach \m in {-3,...,3} \node[sub,anchor=north] at ({8.0+\dm*\m},{9.4-2.5*\dk-.08}) {$\m$};
\node[sub,anchor=west] at ({8.0+3.5*\dm+.08},{9.4-2.5*\dk-.2}) {$m$};

\draw[fine] ({12.7-3.5*\dm-2.5*\sh},{9.4-2.5*\dk}) -- ({12.7+3.5*\dm-2.5*\sh},{9.4-2.5*\dk}) -- ({12.7+3.5*\dm+2.5*\sh},{9.4+2.5*\dk}) -- ({12.7-3.5*\dm+2.5*\sh},{9.4+2.5*\dk}) -- cycle;
