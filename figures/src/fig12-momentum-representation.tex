% Figure 12, momentum representation. At the top left the K ladder as latitude rings on the sphere |J| = sqrt(J(J+1))
% for J = 2, after Harter (1993), fig. 5.5.3, with no energies. At the top right the lattice of torsional labels (m, K)
% in the periodic frame and sheared in the twisted frame, kappa = m + rho K with rho = 1/2, the symmetric frame of
% Mellor, Yurchenko, Mant and Jensen. Below, the periodic torsion modes as radial plots on the torsion circle sort into
% A1, A2 and E pairs, with level rules after fig. 4.2.3. The kappa rows and the species are computed (figures/data).
% Gold marks the shear line, red the versions.
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{numbers.tex}
\begin{document}
\begin{tikzpicture}[plate]
\path[use as bounding box] (0,1.1) rectangle (15,11.9);
% ------------------------------------------------ A. the K ladder on the sphere |J| = sqrt(J(J+1)), J = 2
\begin{scope}[shift={(3.0,9.25)}]
  % the sphere is seen from a higher elevation than the tubes (E = .6) so that the ring planes open and a vector from the
  % center to a ring reads in depth; the body axis stays upright
  \pgfmathsetmacro{\s}{.56}\pgfmathsetmacro{\R}{sqrt(6)*\s}\pgfmathsetmacro{\E}{.6}
  \hatom{0}{0}{\R}{white}
  \draw[fine] (0,{-\R-.25})--(0,{\R+.45}); \node[sub,anchor=south] at (0,{\R+.55}) {body axis $z$};
  \foreach \K in {2,1,0,-1,-2} {\pgfmathsetmacro{\r}{sqrt(6-\K*\K)*\s}\pgfmathsetmacro{\h}{\K*\s*sqrt(1-\E*\E)}
    % a ring point at angle a is visible when sin a < tan(latitude) tan(elevation): K = 2 lies wholly in front, K = -2
    % wholly behind, K = +-1 switch 19.6 degrees from the ends, the equator at its ends
    \pgfmathsetmacro{\vc}{max(-1,min(1,\K*\E/sqrt(1-\E*\E)/sqrt(6-\K*\K)))}\pgfmathsetmacro{\va}{asin(\vc)}
    \ifdim\vc pt<1pt \draw[fine,back] ({\r*cos(\va)},{\h+\E*\r*sin(\va)}) arc[start angle=\va,end angle={180-\va},x radius=\r,y radius={\E*\r}];\fi
    \ifdim\vc pt>-1pt \draw[fine] ({\r*cos(180-\va)},{\h+\E*\r*sin(180-\va)}) arc[start angle={180-\va},end angle={360+\va},x radius=\r,y radius={\E*\r}];\fi
    \node[sub,anchor=east] at ({-\R-.16},\h) {$K=\K$};}
  % the vector J: from the center (marked) up to the right end of the K = 1 ring (its z component is K = 1), named
  % outside the sphere at the right, clear of the lattice beside it
  \pgfmathsetmacro{\ja}{355}\pgfmathsetmacro{\jr}{sqrt(5)*\s}\pgfmathsetmacro{\jh}{\s*sqrt(1-\E*\E)}
  \coordinate (jt) at ({\jr*cos(\ja)},{\jh+\E*\jr*sin(\ja)});
  \draw[harrow] (0,0)--(jt); \inkdot{0,0}
  \node[lab,anchor=west,inner sep=1pt] at ({\R+.1},{\jh+\E*\jr*sin(\ja)+.06}) {$\mathbf J$};
  \node[sub,anchor=north] at (0,{-\R-.4}) {$J=2$, $|\mathbf J|=\sqrt{J(J+1)}\,\hbar$};
\end{scope}
% ------------------------------------------------ B. the (m, K) lattice in the periodic frame and sheared in the twisted frame
\node[sub,anchor=base] at (7.6,11.3) {periodic frame};
\node[sub,anchor=base] at (12.4,11.3) {twisted frame, $\rho=\tfrac12$};
\pgfmathsetmacro{\dm}{.5}\pgfmathsetmacro{\dk}{.5}
\draw[fine] ({7.6-3.5*\dm},{9.4-2.5*\dk}) rectangle ({7.6+3.5*\dm},{9.4+2.5*\dk});
\foreach \K in {-2,...,2} {\pgfmathsetmacro{\y}{9.4+\dk*\K}\node[sub,anchor=east] at ({7.6-3.5*\dm-.1},\y) {$\K$};
  \foreach \m in {-3,...,3} {\inkdot{{7.6+\dm*\m},\y}}}
\foreach \m in {-3,...,3} \node[sub,anchor=north] at ({7.6+\dm*\m},{9.4-2.5*\dk-.08}) {$\m$};
\node[sub,anchor=west] at ({7.6+3.5*\dm+.08},{9.4-2.5*\dk-.2}) {$m$};
\node[sub,anchor=east] at ({7.6-3.5*\dm-.1},{9.4+2.5*\dk+.22}) {$K$};
\pgfmathsetmacro{\sh}{\dm*\rhoIll}
\draw[fine] ({12.4-3.5*\dm-2.5*\sh},{9.4-2.5*\dk}) -- ({12.4+3.5*\dm-2.5*\sh},{9.4-2.5*\dk}) -- ({12.4+3.5*\dm+2.5*\sh},{9.4+2.5*\dk}) -- ({12.4-3.5*\dm+2.5*\sh},{9.4+2.5*\dk}) -- cycle;
% the zero line (ink, thin) and the gold line kappa = rho K, drawn before the lattice dots so that no dot is covered
\draw[thin] (12.4,{9.4-2.5*\dk-.05})--(12.4,{9.4+2.5*\dk+.05});
\draw[line width=1.6pt,draw=ColRetained] ({12.4-2.5*\sh},{9.4-2.5*\dk})--({12.4+2.5*\sh},{9.4+2.5*\dk});
\foreach \K/\list in \kappaRows {\pgfmathsetmacro{\y}{9.4+\dk*\K}\foreach \k in \list {\inkdot{{12.4+\dm*\k},\y}}}
\node[sub,anchor=south west] at ({12.4+2.5*\sh+.05},{9.4+2.5*\dk+.16}) {$\kappa=\rho K$};
\node[sub,anchor=north] at (12.4,{9.4-2.5*\dk-.08}) {$0$}; \node[sub,anchor=north] at ({12.4+3.5*\dm-2.5*\sh+.2},{9.4-2.5*\dk-.08}) {$\kappa$};
\node[sub,anchor=north] at (12.4,7.58) {$\kappa=m+\rho K$};
% ------------------------------------------------ C. periodic torsion modes on the torsion circle, and their species under G6
\node[sub,anchor=north west] at (.337,6.1) {periodic torsion modes on the torsion circle};
\node[sub,anchor=east] at (1.55,4.4) {$\cos m\tau$}; \node[sub,anchor=east] at (1.55,3.2) {$\sin m\tau$}; \node[sub,anchor=base east] at (1.55,5.12) {$m$};
\def\spE{E}\def\spnone{none}
\foreach \m/\cs/\ss in \torsionSpecies {   % the species of cos m tau and sin m tau under G6 (none: sin 0 vanishes)
  \pgfmathsetmacro{\cx}{2.55+1.78*\m}
  \node[lab,anchor=base] at (\cx,5.12) {$\m$};
  \foreach \yc/\f/\sp in {4.4/cos/\cs,3.2/sin/\ss} {
    \draw[thin] (\cx,\yc) circle[radius=.4];   % the torsion circle thin, the mode curve heavy
    \edef\spx{\sp}\ifx\spx\spnone\else \draw[heavy] plot[domain=0:360,samples=180,variable=\t] ({\cx+(.4+.16*\f(\m*\t))*cos(\t+90)},{\yc+(.4+.16*\f(\m*\t))*sin(\t+90)});\fi
    \foreach \a in {90,210,330} {\vpoint{{\cx+.4*cos(\a)},{\yc+.4*sin(\a)}}}}   % the version points after the curve, on top
  \edef\spc{\cs}\edef\sps{\ss}
  \ifx\spc\spE
    \draw[heavy] (\cx-.6,2.425)--(\cx+.6,2.425) (\cx-.6,2.335)--(\cx+.6,2.335); \node[lab,anchor=north] at (\cx,2.26) {$\mathrm E$};
  \else\ifx\sps\spnone
    \draw[heavy] (\cx-.6,2.38)--(\cx+.6,2.38); \node[lab,anchor=north] at (\cx,2.26) {$\mathrm A_1$};
  \else
    \draw[heavy] (\cx-.6,2.38)--(\cx+.6,2.38); \node[lab,anchor=north] at (\cx,2.26) {$\mathrm A_1$ (cos)};
    \draw[heavy] (\cx-.6,1.66)--(\cx+.6,1.66); \node[lab,anchor=north] at (\cx,1.54) {$\mathrm A_2$ (sin)};
  \fi\fi
}
\end{tikzpicture}
