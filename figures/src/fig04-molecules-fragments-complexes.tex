% Figure 4. Molecules, fragments, complexes: the reaction KRb + KRb -> K2Rb2 -> K2 + Rb2 as groupings of one set
% of four nuclei and as a path in the subgroup lattice of S. What the reader must see: a grouping is a choice of
% boundaries on the same configuration; each grouping has its channel group; the reaction climbs from the incoming
% channel group to the complex group S and descends to one of the two product channel groups; the three channel
% groups meet in <E*>; any two generate S. Drawing: the three groupings with grouping lines coloured by what they
% join (potassium green, rubidium violet, mixed in ink), the sixteen subgroups of S with the path in the same two
% colours side by side until they split at S. K2Rb2 positions are schematic (declared); the lattice, the orders,
% the core and the generation facts are computed (compute/groups.py, checks/verify.g). Hatch unused; gold unused.

% ------------------------------------------------ A. one set of four nuclei, grouped three ways along the reaction
\foreach \x/\pname/\cap/\grp in {2.6/in/{$\mathrm{KRb}+\mathrm{KRb}$}/{$G_{\mathrm{in}}=\langle(12)(34),E^*\rangle$},7.5/complex/{$\mathrm K_2\mathrm{Rb}_2$}/{$S=\langle(12),(34),E^*\rangle$},12.4/out/{$\mathrm K_2+\mathrm{Rb}_2$}/{$G_{\mathrm K}=\langle(12),E^*\rangle$,\ $G_{\mathrm{Rb}}=\langle(34),E^*\rangle$}} {

  \end{scope}
}
