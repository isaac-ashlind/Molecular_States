% Figure 8. Nuclear spin weights. What the reader must see: a spatial level of species Gamma admits the spin states
% of the species whose product with Gamma contains the parity character; the count is the weight. Drawing: two rows
% of dot stacks, even parity above and odd below, one column per spatial species, each stack tagged by the spin
% species it draws on, so the swap of A1 and A2 between the rows and the counts 12, 4, 8 are seen; a double rule
% marks the two-dimensional E. Level rules after Harter (1993) fig. 4.2.3 in spirit. Colours: ink only. Hatch unused.
% All counts from the character computation (checks/verify.g).
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{numbers.tex}
\begin{document}
\begin{tikzpicture}[plate]
\path[use as bounding box] (0,0) rectangle (15,8.2);
\newcommand\stack[5]{% x, y, weight, spin tag, double rule flag
  \begin{scope}[shift={(#1,#2)}]
    \draw[heavy] (-1.0,0)--(1.0,0); \ifnum#5=1 \draw[heavy] (-1.0,.06)--(1.0,.06); \fi
    \pgfmathtruncatemacro{\n}{#3}
    \foreach \k in {1,...,\n} {\pgfmathtruncatemacro{\row}{(\k-1)/4}\pgfmathtruncatemacro{\col}{mod(\k-1,4)}
      \fill[PlateInk] ({-.6+.4*\col},{.32+.38*\row}) circle[radius=2.2pt];}
    \node[lab,anchor=west] at (1.15,{.32+.38*((\n-1)/4)/2}) {#3, spin #4};
  \end{scope}}
\foreach \x/\l in {3.0/{\mathrm A_1},7.5/{\mathrm A_2},12.0/{\mathrm E}} {\node[lab,anchor=south] at (\x,7.45) {$\l$};}
\node[sub,anchor=south] at (7.5,7.9) {spatial species};
\node[lab,anchor=east] at (1.7,5.5) {even, $\chi_+$};
\node[lab,anchor=east] at (1.7,2.5) {odd, $\chi_-$};
\stack{3.0}{4.9}{12}{$\mathrm A_1$}{0} \stack{7.5}{4.9}{4}{$\mathrm A_2$}{0} \stack{12.0}{4.9}{8}{$\mathrm E$}{1}
\stack{3.0}{1.9}{4}{$\mathrm A_2$}{0} \stack{7.5}{1.9}{12}{$\mathrm A_1$}{0} \stack{12.0}{1.9}{8}{$\mathrm E$}{1}
\node[sub,anchor=north] at (7.5,1.1) {one dot per spin state the level admits; $12+4+2\cdot8=32$};
\end{tikzpicture}
\end{document}
