% Figure 8. Nuclear spin weights. What the reader must see: the 32-dimensional proton spin space splits under G6
% into 12 A1, 4 A2 and 8 E (pairs); a spatial level pairs with the spin species whose product with it contains the
% parity character, so even-parity levels pair straight across and odd-parity levels pair with A1 and A2 swapped;
% the weight of a level is the size of the bundle it is joined to. Drawing: the spin space in the centre as bundles
% of hairlines, one line per state (E as pairs); even-parity spatial species at the left joined straight across;
% odd-parity species at the right joined with one crossing, all joins solid, each ending on a bracket stroke that
% spans its bundle. Correlation diagram in the manner of Harter (1993)
% fig. 4.2.3. Colours: ink only. Hatch unused. Counts from the character computation (checks/verify.g) and the
% fragment product (checks/verify_spin.py).
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{numbers.tex}
\begin{document}
\begin{tikzpicture}[plate,jn/.style={line width=.5pt,draw=PlateInk},jndash/.style={line width=.5pt,draw=PlateInk,dash pattern=on 2.4pt off 1.6pt}]
\path[use as bounding box] (0,0) rectangle (15,8.8);
% heights of the three species rows
\def\yA{6.3}\def\yB{4.3}\def\yE{2.3}
% ---- the spin space: bundles of hairlines, one line per state, E as pairs
\begin{scope}[shift={(7.5,0)}]
  \foreach \k in {0,...,11} {\draw[line width=.35pt,draw=PlateInk] (-1.1,{\yA-.385+.07*\k})--(1.1,{\yA-.385+.07*\k});}
  \foreach \k in {0,...,3} {\draw[line width=.35pt,draw=PlateInk] (-1.1,{\yB-.105+.07*\k})--(1.1,{\yB-.105+.07*\k});}
  \foreach \k in {0,...,7} {\draw[line width=.35pt,draw=PlateInk] (-1.1,{\yE-.49+.14*\k})--(1.1,{\yE-.49+.14*\k}); \draw[line width=.35pt,draw=PlateInk] (-1.1,{\yE-.49+.14*\k+.045})--(1.1,{\yE-.49+.14*\k+.045});}
  \node[sub,anchor=south] at (0,{\yA+.46}) {$\mathrm A_1$, $12$}; \node[sub,anchor=south] at (0,{\yB+.2}) {$\mathrm A_2$, $4$}; \node[sub,anchor=south] at (0,{\yE+.58}) {$\mathrm E$, $8$ pairs};
  \draw[line width=.7pt,draw=PlateInk] (-1.1,{\yA-.4})--(-1.1,{\yA+.4}) (1.1,{\yA-.4})--(1.1,{\yA+.4}) (-1.1,{\yB-.12})--(-1.1,{\yB+.12}) (1.1,{\yB-.12})--(1.1,{\yB+.12}) (-1.1,{\yE-.5})--(-1.1,{\yE+.55}) (1.1,{\yE-.5})--(1.1,{\yE+.55});
  \node[lab,anchor=south] at (0,7.5) {$\spin$, $32$};
  \draw[fine] (-1.3,7.35)--(-1.3,7.2)--(1.3,7.2)--(1.3,7.35);
\end{scope}
% ---- even-parity levels at the left: straight across
\node[lab,anchor=south] at (1.8,7.3) {even, $\chi_+$};
\draw[heavy] (1.0,\yA)--(2.6,\yA); \node[lab,anchor=east] at (.85,\yA) {$\mathrm A_1$};
\draw[heavy] (1.0,\yB)--(2.6,\yB); \node[lab,anchor=east] at (.85,\yB) {$\mathrm A_2$};
\draw[heavy] (1.0,\yE)--(2.6,\yE) (1.0,{\yE+.06})--(2.6,{\yE+.06}); \node[lab,anchor=east] at (.85,{\yE+.03}) {$\mathrm E$};
\draw[jn] (2.6,\yA)--(6.4,\yA); \draw[jn] (2.6,\yB)--(6.4,\yB); \draw[jn] (2.6,{\yE+.03})--(6.4,{\yE+.03});
% ---- odd-parity levels at the right: A1 and A2 swap partners, E keeps E
\node[lab,anchor=south] at (13.2,7.3) {odd, $\chi_-$};
\draw[heavy] (12.4,\yA)--(14.0,\yA); \node[lab,anchor=west] at (14.15,\yA) {$\mathrm A_1$};
\draw[heavy] (12.4,\yB)--(14.0,\yB); \node[lab,anchor=west] at (14.15,\yB) {$\mathrm A_2$};
\draw[heavy] (12.4,\yE)--(14.0,\yE) (12.4,{\yE+.06})--(14.0,{\yE+.06}); \node[lab,anchor=west] at (14.15,{\yE+.03}) {$\mathrm E$};
\draw[jn] (12.4,\yA)--(8.6,\yB); \draw[jn] (12.4,\yB)--(8.6,\yA); \draw[jn] (12.4,{\yE+.03})--(8.6,{\yE+.03});
% ---- the weights, read as the size of the bundle a level is joined to
\end{tikzpicture}
\end{document}
