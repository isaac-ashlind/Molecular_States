% Figure 8. Nuclear spin weights. What the reader must see: the 32-dimensional proton spin space splits under G6
% into 12 A1, 4 A2 and 8 E (pairs); a spatial level pairs with the spin species whose product with it contains the
% parity character, so even-parity levels pair straight across and odd-parity levels pair with A1 and A2 swapped;
% the weight of a level is the size of the bundle it is joined to. Drawing: the spin space in the centre as bundles
% of hairlines, one line per state (E as pairs); even-parity spatial species at the left joined straight across;
% odd-parity species at the right joined with one crossing, all joins solid, each ending at its bundle. Correlation diagram in the manner of Harter (1993)
% fig. 4.2.3. Colours: ink only. Hatch unused. Counts from the character computation (checks/verify.g) and the
% fragment product (checks/verify_spin.py).
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{numbers.tex}
\begin{document}
\begin{tikzpicture}[plate,jn/.style={fine}]
\path[use as bounding box] (0,.55) rectangle (15,6.75);
% heights of the three species rows
\def\yA{5.0}\def\yB{3.4}\def\yE{1.7}
% ---- the spin space: bundles of hairlines, one line per state, E as pairs
\begin{scope}[shift={(7.5,0)}]
  \foreach \k in {0,...,11} {\draw[thin] (-1.1,{\yA-.385+.07*\k})--(1.1,{\yA-.385+.07*\k});}
  \foreach \k in {0,...,3} {\draw[thin] (-1.1,{\yB-.105+.07*\k})--(1.1,{\yB-.105+.07*\k});}
  \foreach \k in {0,...,7} {\draw[thin] (-1.1,{\yE-.4675+.14*\k})--(1.1,{\yE-.4675+.14*\k}); \draw[thin] (-1.1,{\yE-.4675+.14*\k+.045})--(1.1,{\yE-.4675+.14*\k+.045});}
  \node[sub,anchor=south] at (0,{\yA+.5}) {$\mathrm A_1$, $12$}; \node[sub,anchor=south] at (0,{\yB+.22}) {$\mathrm A_2$, $4$}; \node[sub,anchor=south] at (0,{\yE+.6725}) {$\mathrm E$, $8$ pairs};
  \node[lab,anchor=base] at (0,6.3) {$(\mathbb C^2)^{\otimes5}$, $32$};
  \draw[fine] (-1.1,6.18)--(-1.1,6.03)--(1.1,6.03)--(1.1,6.18);
\end{scope}
% ---- even-parity levels at the left: straight across
\node[lab,anchor=base] at (1.8,6.3) {even, $\chi_+$};
\draw[heavy] (1.0,\yA)--(2.6,\yA); \node[lab,anchor=base east] at (.85,{\yA-.1}) {$\mathrm A_1$};
\draw[heavy] (1.0,\yB)--(2.6,\yB); \node[lab,anchor=base east] at (.85,{\yB-.1}) {$\mathrm A_2$};
\draw[heavy] (1.0,\yE)--(2.6,\yE) (1.0,{\yE+.09})--(2.6,{\yE+.09}); \node[lab,anchor=base east] at (.85,{\yE-.055}) {$\mathrm E$};
\draw[jn] (2.6,\yA)--(6.4,\yA); \draw[jn] (2.6,\yB)--(6.4,\yB); \draw[jn] (2.6,{\yE+.045})--(6.4,{\yE+.045});
% ---- odd-parity levels at the right: A1 and A2 swap partners, E keeps E
\node[lab,anchor=base] at (13.2,6.3) {odd, $\chi_-$};
\draw[heavy] (12.4,\yA)--(14.0,\yA); \node[lab,anchor=base west] at (14.15,{\yA-.1}) {$\mathrm A_1$};
\draw[heavy] (12.4,\yB)--(14.0,\yB); \node[lab,anchor=base west] at (14.15,{\yB-.1}) {$\mathrm A_2$};
\draw[heavy] (12.4,\yE)--(14.0,\yE) (12.4,{\yE+.09})--(14.0,{\yE+.09}); \node[lab,anchor=base west] at (14.15,{\yE-.055}) {$\mathrm E$};
\draw[jn] (12.4,\yA)--(8.6,\yB); \draw[jn] (12.4,\yB)--(8.6,\yA); \draw[jn] (12.4,{\yE+.045})--(8.6,{\yE+.045});
% ---- the weights, read as the size of the bundle a level is joined to
\end{tikzpicture}
\end{document}
