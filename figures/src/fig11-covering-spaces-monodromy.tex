% Figure 11, covering spaces and monodromy, after Harter (1993), fig. 5.4.2. Over a patch of the quotient F/G stand
% the sheets in version pairs {gX, gbX}, the X column at the left and the bX column at the right, each headed by its
% configuration. The loop gamma_t lifts by climbing a column, gamma_b by hopping across, so t then b ends one level up
% on the right and b then t two levels up. At the right the six sheets form a graph, where t turns the outer triangle
% one way and the inner the other. The six drawn are the G6 block of the twelve sheets.
% The relations bt = t^2 b and btb = t^-1 and the best-fit residual between X and bX are computed (checks/). Red marks
% the points over [X], pale gold the patch, blue nitrogen.
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{molecules.tex}
\begin{document}
\begin{tikzpicture}[plate,sheet/d=.9,gnode/.style={circle,ring,fill=PlateWhite,inner sep=0pt,minimum size=8.6mm,dense}]   % the base patch is a sheet's shape, so every sheet maps onto it
\path[use as bounding box] (0,0) rectangle (15,11.3);
\begin{scope}[shift={(5.6,.8)}]
  % the base: a patch of the quotient, one sheet's size, both columns over it; the two loops inside it
  \sheet{0}{0}{heavy,fill=ColPaleGold}
  \coordinate (bx) at (.35,.45);
  \draw[tlift] (bx) .. controls (-1.7,.84) and (-1.7,.06) .. (bx); \draw[blift] (bx) .. controls (2.4,.06) and (2.4,.84) .. (bx);
  \node[sub,anchor=north,inner sep=2.5pt] at (-.95,-.2) {$\gamma_t$}; \node[sub,anchor=north,inner sep=2.5pt] at (1.65,-.2) {$\gamma_b$};   % under their loops, symmetric about [X]
  \node[sub,anchor=west,inner sep=1pt] at (2.62,.45) {$\mathcal F/G$};
  \foreach \k/\ga/\gb/\pl in {0/{\X}/{b\X}/H,1/{t\X}/{tb\X}/tH,2/{t^2\X}/{t^2b\X}/t^2H} {
    \pgfmathsetmacro{\y}{1.7+1.8*\k}
    \sheet{-2.3}{\y}{fine,fill=PlateWhite} \sheet{2.3}{\y}{fine,fill=PlateWhite}
    \coordinate (pa\k) at (-1.95,{\y+.45}); \coordinate (pb\k) at (2.65,{\y+.45});
    \node[sub,anchor=west,inner sep=1pt] at (-3.5,{\y+.22}) {$\ga$}; \node[sub,anchor=west,inner sep=1pt] at (1.1,{\y+.22}) {$\gb$};   % both columns named at the same place on their sheets
    \node[lab,anchor=east] at (-4.3,{\y+.45}) {$\pl$};}
  \draw[projection,shorten <=2.3pt,shorten >=2.3pt] (bx)--(pa0); \draw[projection,shorten <=2.3pt,shorten >=2.3pt] (bx)--(pb0);   % the projections, dotted as on the cover
  \draw[tlift] (pa0) .. controls (-3.0,{1.7+.45+.4}) and (-3.0,{1.7+1.8+.45-.4}) .. (pa1);
  \draw[blift] (pa1)--(pb1);
  \draw[blift] (pa0)--(pb0);
  \draw[tlift] (pb0) .. controls (4.5,{1.7+.45+1.0}) and (4.5,{1.7+3.6+.45-1.0}) .. (pb2);   % stays inside the sheets, clear of the middle sheet's corner
  \foreach \k in {0,1,2} {\vpoint{pa\k} \vpoint{pb\k}}
  \vpoint{bx} \leaderlabel[sub]{bx}{0}{-.65}{north}{$[\X]$}   % on the loop names' line
  \node[sub,anchor=south] at ($(pb1)+(0,.62)$) {$t$ then $b$}; \node[sub,anchor=south] at ($(pb2)+(0,.62)$) {$b$ then $t$};
\end{scope}
% the two configurations over [X] head their columns, drawn at the scale of figure 5
\begin{scope}\mollabelsinside\pic[scale=.8] at (3.3,10.0) {mol3d-mla-X}; \pic[scale=.8] at (7.9,10.0) {mol3d-mla-bX};\end{scope}
\node[lab] at (3.3,8.6) {$\X$}; \node[lab] at (7.9,8.6) {$b\X$}; \draw[blift] (4.3,8.6)--(6.9,8.6);
% the six sheets as a graph, for the bookkeeping: t turns the outer triangle one way and the inner the other;
% edges run node to node so every head ends on a node's border
\begin{scope}[shift={(12.6,6.4)}]
  \foreach \k in {0,1,2} {\pgfmathsetmacro{\a}{90+120*\k}\coordinate (o\k) at ({1.9*cos(\a)},{1.9*sin(\a)}); \coordinate (i\k) at ({.71*cos(\a)},{.71*sin(\a)});}
  \foreach \k/\l in {0/{\X},1/{t\X},2/{t^2\X}} \node[gnode] (no\k) at (o\k) {$\l$};
  \foreach \k/\l in {0/{b\X},1/{tb\X},2/{t^2b\X}} \node[gnode] (ni\k) at (i\k) {$\l$};
  \foreach \k/\n in {0/1,1/2,2/0} \draw[tpath] (no\k) to[bend right=24] (no\n);
  \foreach \k/\n in {0/2,2/1,1/0} \draw[tpath] (ni\k) to[bend left=30] (ni\n);
  \foreach \k in {0,1,2} \draw[blift] (no\k)--(ni\k);
\end{scope}
\end{tikzpicture}
