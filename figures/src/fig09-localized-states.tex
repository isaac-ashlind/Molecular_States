% Figure 9, localized states. The torsion circle unrolled, with three version packets in each of three rows of width
% Delta/d = 1/6, 1/3 and 2/3 (Delta the packet width, d the spacing of the versions). The profiles are densities, so
% neighbors cross at c times the peak height, and the region where two overlap is pale gold. At the right of each row
% the A1 share w_A1 = 1/3 + 2c/3 fills a bar, gold for the third it has with no overlap and pale gold for the overlap
% part 2c/3, with the row's exact c below. The change of fill marks the split, and the divider marks w_A1. The share
% curves carry the three rows as points. As the packets narrow the shares tend to 1/3 and 2/3, and as they broaden
% w_A1 tends to 1.
% Declared, the overlap and shares use the planar Gaussian formula, not a torsional potential. The shares come from
% figures/data (checks/verify.py, checks/verify_symbolic.py). Red marks the versions.

% packets on the unrolled torsion line: \packetrow{x}{y}{Delta in degrees}{width cm}{height cm} draws the three
% periodic Gaussians at tau = 0, 120, 240 (-120) over [-180,180], white, their overlap pale gold, the profiles fine
\newcommand\packetrow[5]{\begin{scope}[shift={(#1,#2)},x={#4cm/360},y={#5cm}]
  \foreach \c in {0,120,-120} {\path[fill=PlateWhite] plot[domain=-180:180,samples=121,variable=\t] (\t,{exp(-((\t-\c)/#3)^2/2)+exp(-((\t-\c-360)/#3)^2/2)+exp(-((\t-\c+360)/#3)^2/2)}) -- (180,0) -- (-180,0) -- cycle;}
  % overlap: the largest of the three pairwise minima, the region where at least two profiles overlap
  \path[fill=ColPaleGold] plot[domain=-180:180,samples=181,variable=\t] (\t,{max(min(exp(-((\t)/#3)^2/2)+exp(-((\t-360)/#3)^2/2)+exp(-((\t+360)/#3)^2/2),exp(-((\t-120)/#3)^2/2)+exp(-((\t-480)/#3)^2/2)+exp(-((\t+240)/#3)^2/2)),min(exp(-((\t)/#3)^2/2)+exp(-((\t-360)/#3)^2/2)+exp(-((\t+360)/#3)^2/2),exp(-((\t+120)/#3)^2/2)+exp(-((\t-240)/#3)^2/2)+exp(-((\t+480)/#3)^2/2)),min(exp(-((\t-120)/#3)^2/2)+exp(-((\t-480)/#3)^2/2)+exp(-((\t+240)/#3)^2/2),exp(-((\t+120)/#3)^2/2)+exp(-((\t-240)/#3)^2/2)+exp(-((\t+480)/#3)^2/2)))}) -- (180,0) -- (-180,0) -- cycle;
  \foreach \c in {0,120,-120} {\draw[fine] plot[domain=-180:180,samples=121,variable=\t] (\t,{exp(-((\t-\c)/#3)^2/2)+exp(-((\t-\c-360)/#3)^2/2)+exp(-((\t-\c+360)/#3)^2/2)});}
  \draw[fine] (-180,0)--(180,0);   % the torsion line, after the fills so that it shows under the broad packets too
  \end{scope}
  \foreach \c in {0,120,-120} {\vpoint{{#1+#4*\c/360},#2}}}
