% Figure 9. Localized states. What the reader must see: three version packets stand on the torsion circle; their
% width sets the overlap c, which sets the projection shares; narrow packets are 1/3 A1 and 2/3 E, broad packets
% all A1. Drawing: Gaussian profiles standing on the circle (sigma = 40 degrees of arc, so sigma/d = 1/3 with
% d = 120 degrees), the union filled gold, the crossings showing the overlap; the share curves against sigma/d
% with the drawn ratio marked and the limits as open circles. Colours: gold = packets; blue = A1 share, red dashed
% = E share; t path blue. Hatch unused. Data from figures/data (verified exactly in checks/verify_symbolic.py).
% Chosen from studies/alternatives/explore-modes-packets (approach 2).
\documentclass[10pt,border=1mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{figurestyle}
\input{primitives.tex}
\input{numbers.tex}
\pgfplotsset{every axis/.append style={font=\fontsize{8}{10}\selectfont,line width=.4pt,tick style={line width=.4pt}}}
\begin{document}
\begin{tikzpicture}[plate]
\path[use as bounding box] (0,0) rectangle (15,8.6);
% ------------------------------------------------ packets standing on the torsion circle, sigma/d = 1/3 exactly
\begin{scope}[shift={(3.3,5.1)}]
  \pgfmathsetmacro{\R}{1.3}\pgfmathsetmacro{\A}{1.15}\pgfmathsetmacro{\sg}{40}
  \foreach \a in {90,210,330} {\path[fill=ColRetained] plot[domain=-92:92,samples=70,variable=\t] ({(\R+\A*exp(-(\t/\sg)^2/2))*cos(\a+\t)},{(\R+\A*exp(-(\t/\sg)^2/2))*sin(\a+\t)}) -- plot[domain=92:-92,samples=12,variable=\t] ({\R*cos(\a+\t)},{\R*sin(\a+\t)}) -- cycle;}
  \foreach \a in {90,210,330} {\draw[heavy] plot[domain=-92:92,samples=80,variable=\t] ({(\R+\A*exp(-(\t/\sg)^2/2))*cos(\a+\t)},{(\R+\A*exp(-(\t/\sg)^2/2))*sin(\a+\t)});}
  \draw[fine] (0,0) circle[radius=\R];
  \foreach \a/\l in {90/{g_0},210/{g_1},330/{g_2}} {\hatom{\R*cos(\a)}{\R*sin(\a)}{.07}{ColDot} \node[lab] at ({(\R+.55)*cos(\a)},{(\R+.55)*sin(\a)}) {$\l$};}
  \draw[tpath] ({\R*cos(100)},{\R*sin(100)}) arc[start angle=100,end angle=200,radius=\R];
  \node[lab,anchor=south east] at ({(\R+.08)*cos(150)},{(\R+.08)*sin(150)}) {$t$};
  \node[sub,fill=white,inner sep=1.5pt] at ({\R*cos(270)},{\R*sin(270)}) {$\tau$};
  \draw[fine] ({(\R+\A+.15)*cos(90)},{(\R+\A+.15)*sin(90)}) arc[start angle=90,end angle=130,radius={\R+\A+.15}]; \node[sub,anchor=south east,inner sep=1pt] at ({(\R+\A+.2)*cos(112)},{(\R+\A+.2)*sin(112)}) {$\sigma$};
  \node[sub,align=center,text width=6.2cm,anchor=north] at (0,{-\R-1.35}) {$g_j(X)=g_0(t^{-j}X)$ standing on the torsion circle at $\tau=0,\,2\pi/3,\,4\pi/3$; profiles drawn with $\sigma/d=1/3$, the crossings show the overlap $c$};
\end{scope}
% ------------------------------------------------ shares against sigma/d
\begin{axis}[at={(7.6cm,2.2cm)},width=7.0cm,height=5.4cm,xmin=0,xmax=1.2,ymin=0,ymax=1.05,
  xlabel={$\sigma/d$},ylabel={share},xlabel style={at={(axis description cs:1,0)},anchor=west,xshift=3pt},ylabel style={at={(0,1)},anchor=south east,rotate=-90,xshift=4pt},
  ytick={0,0.3333,0.6667,1},yticklabels={$0$,$1/3$,$2/3$,$1$},xtick={0,0.3333,0.6667,1},xticklabels={$0$,$1/3$,$2/3$,$1$},
  axis lines=left,axis line style={-},tick align=outside,clip=false]
  \addplot[ColTorsion,line width=.9pt] table[x=x,y=wA1] {gaussian-shares.dat};
  \addplot[ColUmbrella,line width=.9pt,densely dashed] table[x=x,y=wE] {gaussian-shares.dat};
  \addplot[only marks,mark=o,mark size=2pt,PlateInk] coordinates {(0,0.3333) (0,0.6667)};
  \addplot[only marks,mark=*,mark size=1.8pt,PlateInk] coordinates {(0.3333,\shareShownA) (0.3333,\shareShownE)};
  \node[lab,anchor=north west] at (axis cs:1.0,0.9) {$w_{\mathrm A_1}$}; \node[lab,anchor=south west] at (axis cs:1.0,0.1) {$w_{\mathrm E}$};
  \draw[fine,dash pattern=on .6pt off 1.3pt] (axis cs:0.3333,0)--(axis cs:0.3333,\shareShownA); \node[sub,anchor=west] at (axis cs:0.37,0.5) {as drawn};
\end{axis}
\node[sub,align=center] at (11.0,1.0) {$c=\langle g_0,g_1\rangle=e^{-d^2/8\sigma^2}$,\; $w_{\mathrm A_1}=\tfrac{1+2c}{3}$,\; $w_{\mathrm E}=\tfrac{2(1-c)}{3}$,\; $w_{\mathrm A_2}=0$};
\end{tikzpicture}
