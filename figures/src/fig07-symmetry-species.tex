% Figure 7. Symmetry species. What the reader must see: the three version states are three sites on the torsion
% circle; t turns the sites by a third, b is the mirror through H; a vector on the sites is drawn as three discs
% (area = amplitude, ink positive, white negative, a dash for zero); the grid shows each mode with its t image and
% its b image, so A1 reads as unchanged and E as a pair that mixes; beneath, the E plane with the three images of
% v2 at a third turn and the mirror axis, the same geometry as the sites. Mode pictures after Harter (1993)
% fig. 4.4.3 in spirit. Colours: red = the version sites; ink elsewhere; dotted = the starred b. Vectors verified exactly.

% \modepic{x}{y}{R}{a}{b}{c}{style}  style: circle (torsion circle) or tri (triangle)
\newcommand\modepic[7]{\begin{scope}[shift={(#1,#2)}]
  \ifnum#7=1 \draw[fine] (0,0) circle[radius=#3]; \else \draw[fine] (90:#3)--(210:#3)--(330:#3)--cycle; \fi
  \foreach \mang/\mc in {90/#4,210/#5,330/#6} {
    \pgfmathsetmacro{\r}{.42*#3*sqrt(abs(\mc))}\pgfmathsetmacro{\pos}{\mc>0?1:0}
    \ifdim\r cm>0.03cm \ifnum\pos=1 \fill[PlateInk] (\mang:#3) circle[radius=\r]; \else \draw[fine,fill=white] (\mang:#3) circle[radius=\r]; \fi
    \else \draw[fine] ($(\mang:#3)+(-.07,0)$)--($(\mang:#3)+(.07,0)$); \fi}
  \end{scope}}
