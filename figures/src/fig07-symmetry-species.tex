% Figure 7, symmetry species. The three version states are three sites on the torsion circle, t turns them by a
% third and b mirrors them through H. A vector on the sites is three discs (area the amplitude, ink positive, white
% negative, a tick for zero). The grid shows each mode with its t image and its b image, so A1 stays and the E pair
% mixes. Below is the E plane with the three images of v2 under t and the mirror axis, upright under the key.
% The mode pictures follow Harter (1993), fig. 4.4.3. The vectors are exact (checks/verify_symbolic.py). Red marks the
% version sites.

% \modepic{x}{y}{R}{a}{b}{c}: the amplitudes a, b, c at the sites 90, 210, 330 on a torsion circle of radius R
\newcommand\modepic[6]{\begin{scope}[shift={(#1,#2)}]
  \draw[fine] (0,0) circle[radius=#3];
  \foreach \mang/\mc in {90/#4,210/#5,330/#6} {
    \pgfmathsetmacro{\r}{.42*#3*sqrt(abs(\mc))}\pgfmathsetmacro{\pos}{\mc>0?1:0}
    \ifdim\r cm>0.03cm \ifnum\pos=1 \fill[PlateInk] (\mang:#3) circle[radius=\r]; \else \draw[fine,fill=white] (\mang:#3) circle[radius=\r]; \fi
    \else \draw[fine] (\mang:{#3-.09})--(\mang:{#3+.09}); \fi}
  \end{scope}}
% amplitudes: v1=(1,1,1)/sqrt3; v2=(2,-1,-1)/sqrt6; v3=(0,1,-1)/sqrt2; t(a,b,c)=(c,a,b); b(a,b,c)=(a,c,b)
\pgfmathsetmacro{\sA}{1/sqrt(3)}\pgfmathsetmacro{\sB}{2/sqrt(6)}\pgfmathsetmacro{\sC}{1/sqrt(6)}\pgfmathsetmacro{\sD}{1/sqrt(2)}
